{{/* Preserve HugoBlox article metadata and credit every author in publication order. */}} {{- $page := .page -}} {{- $schema := "Article" -}} {{- if eq $page.Type "blog" -}}{{- $schema = "BlogPosting" -}}{{- end -}} {{- $branding := partialCached "functions/get_branding" $page "branding" -}} {{- $authors := slice -}} {{- range $page.Params.authors -}} {{- $profile := partial "functions/get_author_profile" . -}} {{- $author := dict "@type" "Person" "name" $profile.title -}} {{- if $profile.is_owner -}} {{- $author = merge $author (dict "url" site.Home.Permalink) -}} {{- else if (partial "functions/show_author_profile" $profile) -}} {{- with site.GetPage (printf "/authors/%s" $profile.slug) -}} {{- $author = merge $author (dict "url" .Permalink) -}} {{- end -}} {{- end -}} {{- $authors = $authors | append $author -}} {{- end -}} {{- if not $authors -}} {{- $authors = slice (dict "@type" "Person" "name" (partial "functions/get_author_name" $page)) -}} {{- end -}} {{- $article := dict "@context" "https://schema.org" "@type" $schema "mainEntityOfPage" (dict "@type" "WebPage" "@id" $page.Permalink) "headline" $page.Title "datePublished" ($page.PublishDate.Format "2006-01-02T15:04:05Z07:00") "dateModified" ($page.Lastmod.Format "2006-01-02T15:04:05Z07:00") "author" $authors "publisher" (dict "@type" "Organization" "name" $branding.organization "logo" (dict "@type" "ImageObject" "url" (partial "functions/get_logo_url" $page))) "description" .summary -}} {{- with partial "functions/get_featured_image" $page -}} {{- $article = merge $article (dict "image" (slice .Permalink)) -}} {{- end -}}
